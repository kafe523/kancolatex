\documentclass{article}

\usepackage{fontspec}
\setmainfont{Noto Sans CJK TC} % Prepare your own CJK font for Kanji display

\directlua{
    kancolatex = require("kancolatex")
    kancolatex.InitLoadLib("libkancolatex.dll")
    kctx = kancolatex.InitContext("noro_example.json") 
    kancolatex.InitLuaTeXMacro(kctx)

    token.set_macro("customShipAname", kancolatex.AccessPattern(kctx, "CalcManager.FleetInfo.Fleets.0.Ships.0.Data.Name"))
    token.set_macro("customFleetAlosA", kancolatex.AccessPattern(kctx, "PreCalcMap.fleetAlosA"))
    token.set_macro("customFleetAlosB", kancolatex.AccessMacro(kctx, "fleetAlosB"))
}

%
